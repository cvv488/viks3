# CONFIG
- "port": "45000",
- "max_connections": 10000,
- "wait_reg":10000,
- "timeout": 10000,

-  "logmode": "1,j"
    - [0]: 1-лог в файл, 2-лог в консоль, иначе без настройки
    - [1]: меняет уровень логирования от Debug (0-DEBUG 1-INFO 2-WARN 3-ERROR 4-FATAL 5-PANIC 6-NO -1-TRACE)
    - [2]: "j"-лог в json-формате

- "logdir": "d:/WORK/logs",
    создает файл логов app.log по этому пути, если не указан - пишет в консоль

- "keep_alive_timeout": 20000,
- "debug1": 1
